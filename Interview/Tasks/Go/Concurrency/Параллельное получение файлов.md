
## Задача

По легенде `GetFile`это имитация другого сервиса. Изменять или оптимизировать функцию `GetFile` нельзя.
Функция `GetFiles` получает список имён файлов и последовательно вызывает `GetFile` для каждого из них. Необходимо оптимизировать `GetFiles`, распараллелив получение файлов.
Если при получении одного из файлов возникает ошибка, необходимо прекратить дальнейшую обработку и вернуть ошибку.

## Требования

- Функцию `GetFile` изменять нельзя.
- Результат должен содержать полученные файлы в формате `map[string][]byte`.
- При возникновении ошибки обработка остальных файлов должна быть прекращена.
- `GetFiles` должна вернуть возникшую ошибку.

## Исходный код

```go
package main

import (
	"context"
	"crypto/rand"
	"fmt"
	"log"
	"time"
)

func main() {
	start := time.Now()

	m, err := GetFiles(context.TODO(), "1", "2", "3", "4", "5")
	if err != nil {
		log.Fatalln(err)
	}

	fmt.Println(m)
	fmt.Println(time.Since(start))
}

// GetFiles — пример функции, которую нужно оптимизировать.
func GetFiles(
	ctx context.Context,
	names ...string,
) (result map[string][]byte, err error) {
	if len(names) == 0 {
		return nil, nil
	}

	result = make(map[string][]byte, len(names))

	for _, name := range names {
		result[name], err = GetFile(ctx, name)
		if err != nil {
			return nil, err
		}
	}

	return result, nil
}

// GetFile имитирует обращение к другому сервису.
// Изменять или оптимизировать эту функцию не требуется.
func GetFile(ctx context.Context, name string) ([]byte, error) {
	if name == "" {
		return nil, fmt.Errorf("invalid name %q", name)
	}

	ticker := time.NewTicker(time.Second)
	defer ticker.Stop()

	select {
	case <-ctx.Done():
		return nil, ctx.Err()
	case <-ticker.C:
	}

	if name == "invalid" {
		return nil, fmt.Errorf("invalid name %q", name)
	}

	b := make([]byte, 10)

	n, err := rand.Read(b)
	if err != nil {
		return nil, fmt.Errorf("getting file %q: %w", name, err)
	}

	return b[:n], nil
}
```
